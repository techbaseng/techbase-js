let questions=[],qi=0,score=0,correct=0,wrong=0,streak=0,bestStreak=0;
let timer,timeLeft,answered=false;
const history=[];

function shuffle(a){const b=[...a];for(let i=b.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[b[i],b[j]]=[b[j],b[i]];}return b;}

function startQuiz(){
  questions=shuffle(QUESTIONS).slice(0,10);
  qi=0;score=0;correct=0;wrong=0;streak=0;bestStreak=0;history.length=0;
  document.getElementById('quiz-screen').style.display='block';
  document.getElementById('results-screen').style.display='none';
  showQuestion();
}

function showQuestion(){
  answered=false;
  clearInterval(timer);timeLeft=20;updateTimer();
  timer=setInterval(()=>{timeLeft--;updateTimer();if(timeLeft<=0){clearInterval(timer);autoSubmit();}},1000);
  const q=questions[qi];
  document.getElementById('q-counter').textContent=`Q ${qi+1}/10`;
  document.getElementById('q-topic').textContent=q.topic;
  document.getElementById('q-text').textContent=q.q;
  document.getElementById('q-progress').style.width=((qi/10)*100)+'%';
  document.getElementById('score-live').textContent='Score: '+score;
  document.getElementById('next-btn').style.display='none';
  document.getElementById('q-feedback').className='feedback';
  document.getElementById('q-options').innerHTML=q.opts.map((o,i)=>
    `<button class="option" onclick="pickAnswer(${i})">${o}</button>`
  ).join('');
}

function updateTimer(){
  const el=document.getElementById('timer');
  el.textContent='⏱ '+timeLeft;
  el.style.color=timeLeft<=5?'#dc2626':'#d97706';
}

function autoSubmit(){
  if(!answered) pickAnswer(-1);
}

function pickAnswer(chosen){
  if(answered) return;
  answered=true;
  clearInterval(timer);
  const q=questions[qi];
  const correct_idx=q.ans;
  const opts=document.querySelectorAll('.option');
  opts.forEach((o,i)=>{
    o.classList.add('disabled');
    if(i===correct_idx) o.classList.add('correct');
    else if(i===chosen) o.classList.add('wrong');
  });
  const ok=chosen===correct_idx;
  if(ok){score+=timeLeft>10?20:10;correct++;streak++;bestStreak=Math.max(bestStreak,streak);}
  else{wrong++;streak=0;}
  history.push({topic:q.topic,q:q.q,ok,chosen:chosen>=0?q.opts[chosen]:'(timed out)',ans:q.opts[correct_idx]});
  const fb=document.getElementById('q-feedback');
  fb.textContent=ok?'✅ Correct! '+(timeLeft>10?'+20 (fast bonus)':'+10'):
    (chosen===-1?'⏰ Time up! Answer was: '+q.opts[correct_idx]:'❌ Wrong. Answer: '+q.opts[correct_idx]);
  fb.className='feedback show '+(ok?'ok':'bad');
  document.getElementById('next-btn').style.display='block';
}

function nextQuestion(){
  qi++;
  if(qi>=10) showResults();
  else showQuestion();
}

function showResults(){
  clearInterval(timer);
  document.getElementById('quiz-screen').style.display='none';
  document.getElementById('results-screen').style.display='block';
  const pct=Math.round(score/200*100);

  document.getElementById('r-wrong').textContent=wrong;
  document.getElementById('r-pct').textContent=pct+'%';
  document.getElementById('r-streak').textContent=bestStreak;

'+

');
}
