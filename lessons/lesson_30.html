Phase 1 of 2
Math methods reference
📖 Concept

The Math object has static properties and methods. Key: Math.PI, Math.E. Rounding: round, floor, ceil, trunc. Powers: pow, sqrt, cbrt. Range: max, min, abs, sign (-1,0,1). Log: log, log2, log10.

script.js
// Constants
Math.PI;      // 3.14159...
Math.E;       // 2.71828... (Euler's number)
Math.SQRT2;   // 1.41421...

// Rounding
Math.round(4.5);  // 5
Math.floor(4.9);  // 4
Math.ceil(4.1);   // 5
Math.trunc(-4.9); // -4 (toward zero)

// Powers
Math.pow(2, 8);   // 256
Math.sqrt(144);   // 12
Math.cbrt(27);    // 3
Math.hypot(3, 4); // 5 (Pythagorean)

// Min/max/abs
Math.max(3,1,9);  // 9
Math.min(3,1,9);  // 1
Math.max(...[5,2,8,1]); // 8 (spread array)
Math.abs(-42);    // 42
Math.sign(-5);    // -1

// Logarithms
Math.log(Math.E);  // 1
Math.log2(8);      // 3
Math.log10(1000);  // 3

Phase 2 of 2
Math.random() patterns
📖 Concept

Math.random() returns [0, 1). Random integer in range: Math.floor(Math.random() * (max - min + 1)) + min. Pick random array element: arr[Math.floor(Math.random() * arr.length)]. Shuffle: Fisher-Yates algorithm.

script.js
// Random float: 0 <= x < 1
Math.random();

// Random integer 0 to n-1
Math.floor(Math.random() * n);

// Random integer min to max (inclusive)
Math.floor(Math.random() * (max - min + 1)) + min;

// Dice: 1-6
Math.floor(Math.random() * 6) + 1;

// Random array element
const pick = arr => arr[Math.floor(Math.random() * arr.length)];

// Coin flip
Math.random() < 0.5 ? 'heads' : 'tails';

// Weighted random (70% A, 30% B)
Math.random() < 0.7 ? 'A' : 'B';

function mathCalc(){
  const n=parseFloat(document.getElementById('m-in').value)||0;
  const methods=[
    ['Math.round()',Math.round(n)],['Math.floor()',Math.floor(n)],
    ['Math.ceil()',Math.ceil(n)],['Math.trunc()',Math.trunc(n)],
    ['Math.abs()',Math.abs(n)],['Math.sign()',Math.sign(n)],
    ['Math.sqrt()',n>=0?Math.sqrt(n).toFixed(4):'NaN'],
    ['Math.pow(n,2)',Math.pow(n,2)],
    ['Math.log10()',n>0?Math.log10(n).toFixed(4):'NaN'],
    ['Math.PI * n',(Math.PI*n).toFixed(4)],
  ];
  document.getElementById('m-grid').innerHTML=methods.map(([m,v])=>
    `<div style="background:#fef9ee;border:1.5px solid #fde68a;border-radius:8px;padding:8px">
      <div style="font-family:monospace;font-size:11px;color:#d97706;font-weight:700">${m}</div>
      <div style="font-size:1.1rem;font-weight:800;color:#1c1917">${v}</div>
    </div>`).join('');
}
mathCalc();
function genRandom(){
  const min=parseInt(document.getElementById('rnd-min').value)||1;
  const max=parseInt(document.getElementById('rnd-max').value)||100;
  const count=Math.min(parseInt(document.getElementById('rnd-count').value)||10,100);
  const nums=Array.from({length:count},()=>Math.floor(Math.random()*(max-min+1))+min);
  const avg=nums.reduce((a,b)=>a+b,0)/nums.length;
  document.getElementById('rnd-out').textContent=
    'Generated: ['+nums.join(', ')+']
Min: '+Math.min(...nums)+' | Max: '+Math.max(...nums)+' | Avg: '+avg.toFixed(1);
}
let coinFlips={heads:0,tails:0};
function coinFlip(){
  const result=Math.random()<0.5?'heads':'tails';
  coinFlips[result]++;
  const total=coinFlips.heads+coinFlips.tails;
  document.getElementById('random-out').textContent=
    `Result: ${result.toUpperCase()}! 🪙

After ${total} flips:
Heads: ${coinFlips.heads} (${(coinFlips.heads/total*100).toFixed(1)}%)
Tails: ${coinFlips.tails} (${(coinFlips.tails/total*100).toFixed(1)}%)`;
}
const faces=['⚀','⚁','⚂','⚃','⚄','⚅'];
function rollDice(){
  const d1=Math.floor(Math.random()*6)+1;
  const d2=Math.floor(Math.random()*6)+1;
  document.getElementById('random-out').textContent=
    `${faces[d1-1]} + ${faces[d2-1]} = ${d1+d2}

Die 1: ${d1}
Die 2: ${d2}
Total: ${d1+d2}`;
}
