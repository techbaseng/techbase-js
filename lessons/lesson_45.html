Phase 1 of 2
Enabling strict mode
📖 Concept

Add 'use strict'; at the top of a file or function. ES modules and classes are always strict. Strict mode: prevents undeclared variables, disallows duplicate params, makes this undefined in plain calls (not window), prevents deleting variables, and more.

script.js
'use strict';

// Error: undeclared variable
// x = 5; // ❌ ReferenceError in strict mode
let x = 5; // ✅ must declare

// Error: duplicate params
// function f(a, a) {} // ❌ SyntaxError in strict mode

// Error: deleting variables
// delete x; // ❌ SyntaxError

// 'this' in functions
function fn() { return this; }
fn(); // undefined in strict (was window in sloppy)

// Octal literals not allowed
// let n = 010; // ❌ SyntaxError in strict (was 8)
let n = 0o10; // ✅ use ES6 octal

// Writing to read-only properties throws
const obj = Object.freeze({x: 1});
// obj.x = 2; // ❌ TypeError (silently fails in sloppy)
✏️ Your Task

Toggle strict mode on/off in a script. Try assigning to an undeclared variable in both modes. Notice that sloppy mode silently creates a global variable — a nasty bug source!

Phase 2 of 2
Common mistakes strict mode catches
📖 Concept

Strict mode is a tool for writing better code. It prevents: accidental globals (forgetting let), keyword conflicts, with statement, arguments/eval as variable names, future-reserved keywords. Modern projects use ES modules (always strict) or configure linters. Just always write strict code.

script.js
// Without strict: silent bugs
function updateScore() {
  scroe = 94; // typo — creates global 'scroe'!
}
// With strict: ReferenceError — caught immediately!

// with statement (confusing — forbidden in strict)
// with (obj) { x = 1; } // ❌ SyntaxError in strict

// eval/arguments as identifiers
// let eval = 5;      // ❌ strict mode
// let arguments = []; // ❌ strict mode

// Reserved for future (strict only)
// let implements = 1; // ❌
// let interface = 1;  // ❌
// let private = 1;    // ❌

// Always use strict — or use ES modules
// <script type="module"> — always strict!
// class MyClass {} — body always strict!
✏️ Your Task

List 5 bugs that strict mode would have caught in real code. Research 'strict mode JavaScript bugs caught' and document what each error message means.

function strictDemo(type){
  const lines=[];
  switch(type){
    case 'undeclared':
      lines.push('Sloppy mode:');
      lines.push('  score = 94;  // No error!');
      lines.push('  Creates global variable "score" silently.');
      lines.push('  Typos become globals: scroe = 94 → new global "scroe"!');
      lines.push('');
      lines.push("Strict mode ('use strict'):  ");
      lines.push('  score = 94;  // ❌ ReferenceError: score is not defined');
      lines.push('  Bug caught immediately instead of silently causing issues.');
      break;
    case 'this':
      function sloppyFn(){ return typeof this; }
      lines.push('In a plain function call:');
      lines.push('  Sloppy: this = window (global object)');
      lines.push('  Strict: this = undefined');
      lines.push('');
      lines.push('Current context: typeof this = "'+typeof this+'"');
      lines.push('');
      lines.push('Why it matters: accidentally mutating window in sloppy');
      lines.push('mode caused "global pollution" — a common bug source.');
      break;
    case 'freeze':
      const frozen=Object.freeze({x:1,y:2});
      let err='';
      try{frozen.x=99;}catch(e){err=e.message;}
      lines.push('const obj = Object.freeze({x:1, y:2})');
      lines.push('obj.x = 99;');
      lines.push('');
      lines.push('Sloppy mode: silently fails, obj.x stays 1');
      lines.push('Strict mode: ❌ TypeError: '+err);
      lines.push('Current result: obj.x = '+frozen.x);
      break;
    case 'duplicate':
      lines.push('Duplicate parameter names:');
      lines.push('  function f(a, a) { return a; }');
      lines.push('');
      lines.push('Sloppy mode: allowed! Returns last "a" — confusing');
      lines.push('Strict mode: ❌ SyntaxError: Duplicate parameter name');
      lines.push('');
      lines.push('This prevents very confusing bugs in function signatures.');
      break;
  }
  document.getElementById('strict-out').textContent=lines.join('
');
}
