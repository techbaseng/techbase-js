Phase 1 of 2
for loop anatomy
📖 Concept

Three parts: for (init; condition; update). Init runs once. Condition checked before each iteration. Update runs after each iteration. All three parts are optional. Use break to exit early, continue to skip an iteration.

script.js
// Basic for loop
for (let i = 0; i < 5; i++) {
  console.log(i); // 0,1,2,3,4
}

// Reverse
for (let i = 4; i >= 0; i--) {
  console.log(i); // 4,3,2,1,0
}

// Step
for (let i = 0; i <= 20; i += 5) {
  console.log(i); // 0,5,10,15,20
}

// break and continue
for (let i = 0; i < 10; i++) {
  if (i === 3) continue; // skip 3
  if (i === 7) break;    // stop at 7
  console.log(i); // 0,1,2,4,5,6
}

// Nested loops — multiplication table
for (let i = 1; i <= 3; i++) {
  for (let j = 1; j <= 3; j++) {
    console.log(i + ' x ' + j + ' = ' + (i * j));
  }
}

Phase 2 of 2
Array loops and patterns
📖 Concept

Loop over arrays: for(let i=0;i. Cache length: const len=arr.length for performance in hot loops. Loop backwards to safely remove items. Access previous/next element: arr[i-1].

script.js
const scores = [90, 45, 78, 33, 88, 61];

// Loop with index
for (let i = 0; i < scores.length; i++) {
  console.log(i, scores[i]);
}

// Find max manually
let max = scores[0];
for (let i = 1; i < scores.length; i++) {
  if (scores[i] > max) max = scores[i];
}

// Running total
let total = 0;
for (let i = 0; i < scores.length; i++) {
  total += scores[i];
}
const average = total / scores.length;

// Loop backwards (safe removal)
for (let i = scores.length - 1; i >= 0; i--) {
  if (scores[i] < 45) scores.splice(i, 1);
}
