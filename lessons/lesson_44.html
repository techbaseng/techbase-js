Phase 1 of 2
var and function hoisting
📖 Concept

Hoisting: JS moves declarations to the top of their scope BEFORE execution. var declarations are hoisted and initialised to undefined. Function declarations are fully hoisted — you can call them before they appear. let/const are hoisted but NOT initialised (Temporal Dead Zone).

script.js
// var hoisting — declaration moves to top
console.log(myVar); // undefined (not error!)
var myVar = 'hello';
// JS executes it as:
// var myVar;          // hoisted
// console.log(myVar); // undefined
// myVar = 'hello';    // assignment stays

// Function declaration — fully hoisted
greet('Amina');  // ✅ works before definition!
function greet(name) {
  return 'Hello ' + name;
}

// Function expression — NOT hoisted
// sayBye('Chidi');  // ❌ TypeError: sayBye is not a function
const sayBye = function(name) {
  return 'Bye ' + name;
};

// let/const — Temporal Dead Zone
// console.log(myLet); // ❌ ReferenceError (TDZ)
let myLet = 'world';

Phase 2 of 2
Hoisting best practices
📖 Concept

Avoid relying on hoisting — it makes code unpredictable. Best practices: declare all variables at the top of their scope, always use let/const, define functions before calling them. Treat TDZ errors as helpful — they catch bugs var would silently hide.

script.js
// Common hoisting bug
function setup() {
  // var i leaks from the for loop
  for (var i = 0; i < 3; i++) {}
  console.log(i); // 3 — might be unexpected!

  // let i stays in the loop
  for (let j = 0; j < 3; j++) {}
  // console.log(j); // ReferenceError — correct!
}

// Class declarations are hoisted but NOT initialised
// const s = new Student(); // ❌ ReferenceError
class Student {
  constructor(name) { this.name = name; }
}
const s = new Student('Amina'); // ✅

// Arrow function — NOT hoisted
// const result = double(5); // ❌ Cannot access before init
const double = n => n * 2;
const result = double(5); // ✅

function hoistDemo(type){
  const lines=[];
  switch(type){
    case 'var':
      lines.push('What you write:');
      lines.push('  console.log(x);  // line 1');
      lines.push('  var x = 5;       // line 2');
      lines.push('');
      lines.push('What JS executes:');
      lines.push('  var x;           // hoisted to top (undefined)');
      lines.push('  console.log(x);  // undefined');
      lines.push('  x = 5;          // assignment stays');
      break;
    case 'fn':
      lines.push('Function declarations are FULLY hoisted:');
      lines.push('  greet("Amina"); // works! → "Hello Amina"');
      lines.push('  function greet(n){ return "Hello "+n; }');
      lines.push('');
      lines.push('Function EXPRESSIONS are NOT hoisted:');
      lines.push('  sayHi("Tunde"); // ❌ TypeError!');
      lines.push('  const sayHi = n => "Hi "+n;');
      break;
    case 'let':
      lines.push('let/const — Temporal Dead Zone (TDZ):');
      lines.push('  // TDZ starts here for myLet');
      lines.push('  console.log(myLet); // ❌ ReferenceError!');
      lines.push('  let myLet = "hello"; // TDZ ends here');
      lines.push('');
      lines.push('Compare with var:');
      lines.push('  console.log(myVar); // undefined (no error)');
      lines.push('  var myVar = "hello";');
      lines.push('');
      lines.push('TDZ is safer — catches bugs early!');
      break;
