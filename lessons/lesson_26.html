Phase 1 of 2
indexOf, includes and find
📖 Concept

Search arrays with: indexOf(val) — first index (-1 if not found). lastIndexOf(val) — last index. includes(val) — boolean, uses SameValueZero (finds NaN!). find(fn) — first element where fn returns true. findIndex(fn) — index of first match.

script.js
const nums = [10, 20, 30, 20, 40];
const students = [{name:'Amina',score:94},{name:'Chidi',score:67}];

nums.indexOf(20);          // 1 (first)
nums.lastIndexOf(20);      // 3 (last)
nums.indexOf(99);          // -1
nums.includes(30);         // true
nums.includes(NaN);        // false  but:
[NaN].includes(NaN);       // true! (indexOf misses NaN)

// find — returns element
students.find(s => s.score > 80);       // {name:'Amina',score:94}
students.find(s => s.score > 99);       // undefined

// findIndex
students.findIndex(s => s.name === 'Chidi'); // 1

Phase 2 of 2
some, every and flat search
📖 Concept

some(fn) — true if ANY element passes. every(fn) — true if ALL pass. Short-circuit: some stops at first true, every stops at first false. Array.isArray(val) — check if value is an array.

script.js
const scores = [94, 67, 52, 88, 43];

scores.some(s => s >= 90);    // true (94 passes)
scores.every(s => s >= 40);   // true (all >= 40)
scores.every(s => s >= 50);   // false (43 fails)
scores.some(s => s === 100);  // false

// Practical use
const cart = [{qty:2},{qty:0},{qty:5}];
cart.some(item => item.qty === 0);     // true — out of stock!
cart.every(item => item.qty > 0);     // false

Array.isArray([]);         // true
Array.isArray('string');   // false
Array.isArray({length:3}); // false

function runSearch(){
  const t=document.getElementById('s-term').value.trim();
  const minScore=parseInt(t);
  let res;
  if(!isNaN(minScore)){
    res=students.filter(s=>s.score>=minScore);

      `includes("${t}"): ${inc}\nfind(): ${found?JSON.stringify(found):'undefined'}\nfindIndex(): ${idx}`;

}
