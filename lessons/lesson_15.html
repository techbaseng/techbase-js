Phase 1 of 2
Object literals and property access
📖 Concept

Objects store key-value pairs. Create with {}. Access: dot notation obj.name or bracket notation obj['name'] (use for dynamic keys). Add property: obj.new = value. Delete: delete obj.prop. Check existence: 'prop' in obj or obj.prop !== undefined.

script.js
const student = {
  name:   'Amina',
  score:  94,
  course: 'JavaScript',
  active: true,
};

// Dot notation (most common)
console.log(student.name);   // 'Amina'

// Bracket notation (for dynamic keys)
const field = 'score';
console.log(student[field]); // 94

// Add / modify / delete
student.grade = 'A';         // add
student.score = 95;          // modify
delete student.active;       // delete

// Check if property exists
'name' in student;           // true
'age' in student;            // false
student.age !== undefined;   // false (another way)

// Iterate properties
for (const key in student) {
  console.log(key, ':', student[key]);
}

// Get keys, values, entries
Object.keys(student);    // ['name', 'score', 'course', 'grade']
Object.values(student);  // ['Amina', 95, 'JavaScript', 'A']
Object.entries(student); // [['name','Amina'], ...]

Phase 2 of 2
Methods and this
📖 Concept

A method is a function stored as an object property. Inside a method, this refers to the calling object. Arrow functions don't have their own this — they inherit from the surrounding scope. Shorthand method syntax: greet() { } (no function keyword).

script.js
const student = {
  name:  'Amina',
  score: 94,

  // Method shorthand
  greet() {
    return `Hi, I'm ${this.name}!`; // 'this' = student
  },

  // Computed method
  getGrade() {
    if (this.score >= 70) return 'A';
    if (this.score >= 60) return 'B';
    return 'C';
  },
};

console.log(student.greet());    // 'Hi, I'm Amina!'
console.log(student.getGrade()); // 'A'

// Object spread — copy/merge objects
const extra = { course: 'JS', year: 2025 };
const full = { ...student, ...extra };

// Object.assign — alternative to spread
const copy = Object.assign({}, student);

// Computed property names
const field = 'email';
const profile = { [field]: 'amina@test.com' }; // dynamic key

  const contacts = {};
  let nextId = 1;
  function addContact() {
    const name   = document.getElementById('c-name').value.trim();
    const email  = document.getElementById('c-email').value.trim();
    const phone  = document.getElementById('c-phone').value.trim();
    const course = document.getElementById('c-course').value;
    if (!name) return alert('Name is required!');
    const id = 'contact_' + nextId++;
    contacts[id] = {
      id, name, email, phone, course,
      createdAt: new Date().toLocaleDateString(),
      getInitials() { return this.name.split(' ').map(n=>n[0]).join('').toUpperCase(); },

    document.getElementById('c-email').value = '';
  }
  function renderContacts() {
    const ul = document.getElementById('contact-list');
    const ids = Object.keys(contacts);
    if (!ids.length) { ul.innerHTML = '<li class="empty">No contacts yet</li>'; return; }

      </li>`;

    delete contacts[id];
