Phase 1 of 2
Scope chain and lexical scope
📖 Concept

Scope determines where a variable is accessible. Lexical scope: scope is determined by where the code is WRITTEN (not called). Inner functions can access outer variables. Outer functions cannot access inner variables. The engine walks up the scope chain to resolve variables.

script.js
// Global scope
let globalVar = 'I am global';

function outer() {
  let outerVar = 'I am outer';

  function inner() {
    let innerVar = 'I am inner';
    // Can access: innerVar, outerVar, globalVar
    console.log(globalVar);  // ✅
    console.log(outerVar);   // ✅
    console.log(innerVar);   // ✅
  }

  inner();
  // Can access: outerVar, globalVar
  // console.log(innerVar);  // ❌ ReferenceError
}

outer();
// console.log(outerVar);   // ❌ ReferenceError

// Block scope (let/const)
{
  let blockVar = 'block';
}
// console.log(blockVar);  // ❌ ReferenceError (good!)

Phase 2 of 2
IIFE and module scope
📖 Concept

IIFE (Immediately Invoked Function Expression): (function(){...})() or (() => {})(). Creates a private scope — variables inside don't pollute global scope. Used in older code before modules. Today, ES modules (import/export) provide file-level scope automatically.

script.js
// IIFE — creates immediate private scope
(function() {
  const privateVar = 'nobody can see me';
  console.log('IIFE ran:', privateVar);
})(); // <-- immediately invoked

// Arrow IIFE
(() => {
  const x = 42;
  console.log('Arrow IIFE:', x);
})();

// IIFE with return value
const result = (function() {
  const data = [1, 2, 3];
  return data.reduce((a, b) => a + b, 0);
})();
console.log(result); // 6

// Avoid global pollution — old pattern
(function(global) {
  global.myLib = {
    version: '1.0',
    greet: () => 'Hello!',
  };
})(window);
✏️ Your Task

Wrap your utility functions in an IIFE that exposes only a public API object. The internal helpers stay private. Test that the private variables are inaccessible from the console.

function blockScope(){
  const lines=[];
  {
    var varInBlock='var leaks!';
    let letInBlock='let stays';
  }
  lines.push('Inside block: var and let both work');
  lines.push('');
  lines.push('Outside block:');
  lines.push('  varInBlock = "'+varInBlock+'" ← var LEAKED out! 🚨');
  lines.push('  letInBlock = ❌ ReferenceError (stays in block) ✅');
  lines.push('');
  lines.push('Rule: Always use let/const. var leaks from blocks.');
  document.getElementById('block-out').textContent=lines.join('
');
}
function iifeDemo(){
  const result1=(function(){
    const secret='private';
    const nums=[1,2,3,4,5];
    return{sum:nums.reduce((a,b)=>a+b,0),count:nums.length};
  })();
  const result2=((init)=>{
    let count=init;
    return{increment:()=>++count,get:()=>count};
  })(10);
  result2.increment();result2.increment();result2.increment();
  const lines=[
    'IIFE #1 — returns computed value:',
    '  result1.sum   = '+result1.sum,
    '  result1.count = '+result1.count,
    '  "secret" variable is private — inaccessible outside',
    '',
    'IIFE #2 — counter starting at 10, incremented 3 times:',
    '  result2.get() = '+result2.get(),
    '  Internal count is private — cannot be directly accessed',
    '',
    'IIFEs create private scope without polluting global!',
  ];
  document.getElementById('iife-out').textContent=lines.join('
');
}
