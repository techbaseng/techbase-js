Phase 1 of 2
The iterable protocol
📖 Concept

An iterable has a [Symbol.iterator]() method that returns an iterator. An iterator has a next() method returning {value, done}. Built-in iterables: arrays, strings, Maps, Sets, NodeLists, generators. Spread [...iter] and destructuring also use the protocol.

script.js
// Anything with [Symbol.iterator] is iterable
const arr = [1, 2, 3];
const iter = arr[Symbol.iterator]();
iter.next(); // {value:1, done:false}
iter.next(); // {value:2, done:false}
iter.next(); // {value:3, done:false}
iter.next(); // {value:undefined, done:true}

// Strings are iterable
'hi'[Symbol.iterator]; // function exists
[...'hello'];           // ['h','e','l','l','o']

// Maps and Sets
const map = new Map([['a',1],['b',2]]);
[...map];  // [['a',1],['b',2]]

const set = new Set([1,2,3]);
[...set];  // [1,2,3]

// Check if something is iterable
const isIterable = val =>
  val != null && typeof val[Symbol.iterator] === 'function';

isIterable([]);     // true
isIterable('hi');   // true
isIterable({});     // false — plain objects are NOT iterable

Phase 2 of 2
Custom iterable with Symbol.iterator
📖 Concept

Create custom iterables by adding [Symbol.iterator]() to an object. The method returns an object with a next() method. This lets your objects work with for...of, spread, and destructuring. Generator functions are the easiest way to make iterables.

script.js
// Custom iterable — range object
const range = {
  from: 1,
  to:   5,
  [Symbol.iterator]() {
    let current = this.from;
    const last = this.to;
    return {
      next() {
        if (current <= last) {
          return { value: current++, done: false };
        }
        return { value: undefined, done: true };
      }
    };
  }
};

for (const n of range) console.log(n); // 1,2,3,4,5
[...range]; // [1,2,3,4,5]

// Easier with a generator
function* createRange(from, to) {
  for (let i = from; i <= to; i++) yield i;
}
for (const n of createRange(1, 5)) console.log(n);

function inspect(type){
  let iter,lines=[],val;
  switch(type){
    case 'array': val=[10,20,30];break;
    case 'string': val='hello';break;
    case 'map': val=new Map([['a',1],['b',2]]);break;
    case 'set': val=new Set([1,2,3,2,1]);break;
    case 'object': val={a:1,b:2};break;
  }
  const hasIter=val!=null&&typeof val[Symbol.iterator]==='function';
  lines.push(`Type: ${type === 'object'?'plain object':type}`);
  lines.push(`Has [Symbol.iterator]: ${hasIter}`);
  if(hasIter){
    iter=val[Symbol.iterator]();
    lines.push('');
    lines.push('Calling .next() manually:');
    for(let i=0;i<4;i++){
      const r=iter.next();
      lines.push('  .next() → '+JSON.stringify(r));
      if(r.done)break;
    }
    lines.push('');
    lines.push('[...spread]: '+JSON.stringify([...val]).slice(0,60));
  } else {
    lines.push('
❌ Not iterable — cannot use for...of or spread');
  }
  document.getElementById('inspect-out').textContent=lines.join('
');
}
function buildRange(){

length: ${result.length}
sum: ${result.reduce((a,b)=>a+b,0)}
average: ${(result.reduce((a,b)=>a+b,0)/result.length).toFixed(2)}`;
