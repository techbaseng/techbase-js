Phase 1 of 2
Identifiers and keywords
📖 Concept

JavaScript is case sensitive — myVar ≠ myvar. Identifiers (variable/function names) must start with a letter, $ or _. Use camelCase for variables (firstName), PascalCase for classes (StudentRecord), SCREAMING_SNAKE for constants (MAX_SIZE). Keywords are reserved: let const var if else for while function return class import — you cannot use them as variable names.

script.js
// ✅ Valid identifiers
let firstName = 'Tunde';
let _private = 42;
let $price = 9.99;
let camelCase = true;
const MAX_RETRIES = 3;

// ❌ Invalid identifiers
// let 1name = 'bad';   // starts with number
// let my-var = 'bad';  // hyphen not allowed
// let if = 'bad';      // reserved keyword

// Case sensitive!
let colour = 'blue';
let Colour = 'red';   // completely different variable!
console.log(colour);  // blue
console.log(Colour);  // red

// Naming conventions
const MAX_SIZE = 100;         // constant
let studentName = 'Amina';   // variable (camelCase)
function calculateScore() {} // function (camelCase)
class StudentRecord {}       // class (PascalCase)

Phase 2 of 2
Literals and expressions
📖 Concept

A literal is a fixed value written directly in code. Types: numeric (42, 3.14), string ('hello', "world"), boolean (true/false), null, array ([1,2,3]), object ({key:'val'}). An expression produces a value: 2+2, 'hi'.length, arr[0]. A statement performs an action.

script.js
// Literals
42              // number literal
3.14            // float literal
'Hello'         // string literal (single quotes)
"World"         // string literal (double quotes)
`Template`      // template literal (backticks)
true            // boolean literal
null            // null literal
[1, 2, 3]      // array literal
{a: 1, b: 2}  // object literal

// Expressions — produce a value
2 + 2           // 4
'Hello'.length  // 5
arr[0]          // first element
x > 5 ? 'big' : 'small'  // ternary

// Statements — perform an action
let x = 5;              // declaration
if (x > 3) { ... }     // control flow
console.log(x);         // call
✏️ Your Task

In DevTools console, type each literal type and press Enter to see how JS represents it. Then combine literals into expressions: 'Techbase'.length + 3 — what does this produce?

<script>
  const KEYWORDS = ['let','const','var','if','else','for','while','do','switch','case','break','continue','return','function','class','import','export','new','this','typeof','instanceof','try','catch','finally','throw','async','await','yield','delete','in','of','null','true','false','undefined'];
  KEYWORDS.forEach(k => {
    const s = document.createElement('span');
    s.className = 'kw'; s.textContent = k;
    document.getElementById('kw-list').appendChild(s);
  });
  function checkIdent() {
    const val = document.getElementById('ident').value.trim();
    const res = document.getElementById('ident-result');
    if (!val) { res.style.display='none'; return; }
    res.style.display = 'block';
    if (KEYWORDS.includes(val)) {
      res.className = 'result bad'; res.textContent = '❌ Reserved keyword — cannot be used as identifier!';
    } else if (/^[a-zA-Z_$][a-zA-Z0-9_$]*$/.test(val)) {
      const style = /^[A-Z]/.test(val) ? ' (PascalCase — use for classes)' :
                    val === val.toUpperCase() ? ' (SCREAMING_SNAKE — use for constants)' :
                    ' (camelCase — use for variables and functions)';
      res.className = 'result ok'; res.textContent = '✅ Valid identifier!' + style;
    } else {
      res.className = 'result bad';
      const reason = /^\d/.test(val) ? 'Cannot start with a number.' :
                     /[-\s]/.test(val) ? 'Cannot contain hyphens or spaces.' :
                     'Contains invalid characters.';
      res.textContent = '❌ Invalid: ' + reason;
    }
  }
</script>
