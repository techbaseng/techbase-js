Phase 1 of 2
let vs var block scope
📖 Concept

let is block-scoped — it only exists inside the nearest { } block. This makes loops and conditionals predictable. Classic var bug: for (var i=0; i<3; i++) then setTimeout — all callbacks see i=3. With let, each iteration creates a new binding — each callback sees its own i.

script.js
// Classic var bug in loops
for (var i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 0);  // logs: 3 3 3 (not 0 1 2!)
}

// Fixed with let — each iteration has its own i
for (let i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 0);  // logs: 0 1 2 ✅
}

// Block scope in action
if (true) {
  var leaks = 'I escape the block';
  let stays = 'I stay in the block';
}
console.log(leaks);  // works — var escapes!
// console.log(stays);  // ReferenceError — let stays

// Useful pattern: swap variables
let a = 1, b = 2;
[a, b] = [b, a];  // destructuring swap — no temp needed
console.log(a, b); // 2, 1
✏️ Your Task

Run the setTimeout loop example with both var and let in DevTools. See how var gives 3,3,3 while let correctly gives 0,1,2. This is one of the most common JS interview questions!

Phase 2 of 2
Temporal Dead Zone
📖 Concept

let variables are hoisted but NOT initialised — accessing them before declaration throws a ReferenceError (not undefined like var). This 'Temporal Dead Zone' exists from the start of the block until the let declaration. This catches bugs that var would silently hide.

script.js
// TDZ demo
{
  // TDZ starts here for 'name'
  // console.log(name);  // ❌ ReferenceError: Cannot access before init

  let name = 'Amina';  // TDZ ends here

  console.log(name);   // ✅ 'Amina'
}

// Let in for loop — new binding per iteration
const funcs = [];
for (let i = 0; i < 5; i++) {
  funcs.push(() => i);  // captures the CURRENT i each time
}
console.log(funcs.map(f => f())); // [0, 1, 2, 3, 4]

// Can't re-declare in same scope
let city = 'Ibadan';
// let city = 'Lagos';  // ❌ SyntaxError: already declared
✏️ Your Task

Create an array of 5 functions using a for loop with let. Each function should return its loop index multiplied by 10. Call all 5 functions and log the results. Would this work with var? (No — all would return 50!)

    const results = [];
    // Simulating async — collect values after all iterations
    for (var i = 0; i < 3; i++) {
      (function(j){ setTimeout(() => results.push('var: ' + j), 10*j+10); })(i); // IIFE workaround

    const bugResults = [];

  function letDemo() {
    const results = [];
    for (let i = 0; i < 3; i++) {
      const captured = i;
      setTimeout(() => results.push('let captures: ' + captured), 10 * i + 10);
    }
    setTimeout(() => {
      document.getElementById('loop-out').textContent =

  ');

  function scopeDemo() {
    const lines = [];
    if (true) {
      let blockLet = 'block-scoped let';
      var leakyVar = 'function-scoped var';
      lines.push('Inside if block:');
      lines.push('  blockLet = "' + blockLet + '"');
      lines.push('  leakyVar = "' + leakyVar + '"');

    lines.push('Outside if block:');
    lines.push('  blockLet = ❌ ReferenceError (let stays in block)');
    lines.push('  leakyVar = "' + leakyVar + '" ← var escaped!');
    document.getElementById('scope-out').textContent = lines.join('
');
