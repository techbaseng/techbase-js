Phase 1 of 2
Function declarations and expressions
📖 Concept

Declaration: function name() {} — hoisted, can be called before definition. Expression: const f = function() {} — not hoisted. Arrow function: const f = () => {} — shorter, no own this. Parameters have default values: function greet(name='World'). Return a value with return; without return, functions return undefined.

script.js
// Function declaration (hoisted)
greet('Amina');  // ✅ can call before definition
function greet(name) {
  console.log('Hello, ' + name + '!');
}

// Function expression (not hoisted)
const add = function(a, b) {
  return a + b;
};

// Arrow function (concise)
const multiply = (a, b) => a * b;      // implicit return
const square   = n => n * n;           // one param, no parens
const getPI    = () => 3.14159;        // no params

// Default parameters
function createUser(name, role = 'student', score = 0) {
  return { name, role, score };
}
createUser('Tunde');               // {name:'Tunde', role:'student', score:0}
createUser('Amina', 'teacher');    // {name:'Amina', role:'teacher', score:0}

// Rest parameters
function sum(...numbers) {
  return numbers.reduce((total, n) => total + n, 0);
}
sum(1, 2, 3, 4, 5);  // 15

Phase 2 of 2
Pure functions and return values
📖 Concept

A pure function: always returns the same output for the same input, no side effects (doesn't modify external state). Side effects: modifying global variables, DOM manipulation, console.log, API calls. Pure functions are easy to test and reason about. Impure is fine for UI — but separate logic from effects.

script.js
// Pure — same input, same output, no side effects
function add(a, b) { return a + b; }
function toUpperCase(str) { return str.toUpperCase(); }
function getFullName(first, last) { return `${first} ${last}`; }

// Impure — side effects
let total = 0;
function addToTotal(n) {
  total += n;   // modifies external state
  return total;
}

// Multiple return values via destructuring
function minMax(arr) {
  return [Math.min(...arr), Math.max(...arr)];
}
const [min, max] = minMax([3, 1, 7, 2, 9]);

// Functions as arguments (higher-order functions)
const nums = [1, 2, 3, 4, 5];
const doubled = nums.map(n => n * 2);   // [2,4,6,8,10]
const evens   = nums.filter(n => n % 2 === 0); // [2,4]
const total2  = nums.reduce((acc, n) => acc + n, 0); // 15
