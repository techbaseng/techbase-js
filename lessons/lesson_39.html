Use while and do...while loops for condition-driven iteration

📚 2 phases 🏗️ Build: My While Loop Lab
Phase 1 of 2
while loop
📖 Concept

while(condition) — checks condition BEFORE each iteration. If the condition starts false, the body never runs. Always ensure the condition eventually becomes false or use break — infinite loops freeze the browser! Use while when you don't know how many iterations are needed.

script.js
// Basic while
let i = 0;
while (i < 5) {
  console.log(i);
  i++; // MUST update condition variable!
}

// Unknown iterations — reading user input
let attempts = 0;
let password = '';
while (password !== 'secret' && attempts < 3) {
  password = prompt('Enter password:');
  attempts++;
}

// Find first prime > 100
let n = 101;
while (!isPrime(n)) n++;
console.log(n); // 101 (it is prime)

// Loop with break
let count = 0;
while (true) {  // infinite — must break!
  count++;
  if (count >= 10) break;
}

Write a while loop that generates random numbers until it gets one above 0.9. Count how many attempts it took. Run it 5 times and log the average attempts.

Phase 2 of 2
do...while loop
📖 Concept

do { } while(condition) — runs body ONCE first, THEN checks condition. Use when you need at least one execution: input validation, game loops, menu prompts. Rarer than while — only use when the first-run guarantee matters.

script.js
// do...while — body always runs at least once
let num;
do {
  num = parseInt(prompt('Enter a number 1-10:'));
} while (num < 1 || num > 10 || isNaN(num));

// Countdown timer
let count = 5;
do {
  console.log(count);
  count--;
} while (count > 0);
// logs: 5,4,3,2,1

// Difference from while:
let x = 100;
while (x < 10) { console.log('while'); }    // never runs
do { console.log('do-while'); } while (x < 10); // runs once!

let target,guesses=0,gameActive=false;
function startGame(){
  target=Math.floor(Math.random()*100)+1;
  guesses=0; gameActive=true;
  document.getElementById('guess-setup').style.display='none';
  document.getElementById('guess-play').style.display='block';
  document.getElementById('guess-hint').textContent='I picked a number 1–100. Start guessing!';

document.addEventListener('keydown',e=>{if(e.key==='Enter'&&gameActive)makeGuess();});
function makeGuess(){
  const g=parseInt(document.getElementById('guess-in').value);
  if(!g||g<1||g>100){document.getElementById('guess-out').textContent='Enter a number 1–100!';return;}
  guesses++;
  if(g===target){
    document.getElementById('guess-hint').textContent='🎉 Correct!';
    document.getElementById('guess-out').textContent=`You got it in ${guesses} guess${guesses!==1?'es':''}!
The number was ${target}.`;
    gameActive=false;
    document.getElementById('guess-setup').style.display='block';
    document.getElementById('guess-setup').innerHTML='<button class="btn" onclick="startGame()">🎲 Play Again</button>';
  } else {
    const hint=g<target?'📈 Higher!':'📉 Lower!';
    document.getElementById('guess-hint').textContent=hint;
    document.getElementById('guess-out').textContent=`Guess #${guesses}: ${g} — ${hint}`;
  }
  document.getElementById('guess-in').value='';
}
function randomUntil(){
  const thresh=parseFloat(document.getElementById('threshold').value)||0.9;
  const results=[];
  for(let r=0;r<5;r++){
    let tries=0,val=0;
    while(val<=thresh){val=Math.random();tries++;}
    results.push(`Run ${r+1}: ${tries} tries → got ${val.toFixed(4)}`);
  }
  const totalTries=results.map(r=>parseInt(r.split(':')[1])).reduce((a,b)=>a+b,0);
  document.getElementById('rand-out').textContent=results.join('
')+`

Avg tries: ${(totalTries/5).toFixed(1)}
Expected: ~${(1/(1-thresh)).toFixed(1)} (1/(1-threshold))`;
}

  ];

');
