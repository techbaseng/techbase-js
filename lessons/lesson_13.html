Phase 1 of 2
Primitive types
📖 Concept

7 primitive types: string, number, bigint, boolean, undefined, null, symbol. Check with typeof. Primitives are immutable — operations return new values. typeof null === 'object' is a historic bug. undefined = declared but not assigned; null = intentionally empty.

script.js
typeof 42           // "number"
typeof 'hello'      // "string"
typeof true         // "boolean"
typeof undefined    // "undefined"
typeof null         // "object"  ← historic bug!
typeof {}           // "object"
typeof []           // "object"  ← arrays are objects
typeof function(){} // "function"

// Check for null correctly
let x = null;
x === null;          // true ✅
typeof x === 'object' && x === null;  // also works

// undefined vs null
let a;              // undefined (not yet assigned)
let b = null;       // null (intentionally empty)
console.log(a === undefined); // true
console.log(b === null);      // true
console.log(a == b);          // true (loose — both "empty")
console.log(a === b);         // false (different types)

Phase 2 of 2
Type coercion
📖 Concept

JS automatically converts types in certain contexts — this is coercion. String + number = string: '5' + 3 = '53'. Number context: '5' - 3 = 2. Explicit conversion: Number('42'), String(42), Boolean(0). Falsy values: 0, '', null, undefined, NaN, false.

script.js
// Implicit coercion (surprising!)
'5' + 3;      // '53'  (number → string)
'5' - 3;      // 2     (string → number)
'5' * '3';    // 15    (both → number)
true + 1;     // 2
false + 1;    // 1
null + 1;     // 1

// Falsy values (evaluate to false in boolean context)
if (!0)         console.log('0 is falsy');
if (!'')        console.log('empty string is falsy');
if (!null)      console.log('null is falsy');
if (!undefined) console.log('undefined is falsy');
if (!NaN)       console.log('NaN is falsy');
// Everything else is truthy: '0', 'false', [], {}

// Explicit conversion (always prefer this)
Number('42');      // 42
Number('hello');   // NaN
Number(true);      // 1
String(42);        // '42'
Boolean(0);        // false
Boolean('hello');  // true

  function checkType() {
    const raw = document.getElementById('val-in').value.trim();
    const out = document.getElementById('type-out');
    try {
      const val = eval(raw);
      const t = typeof val;
      const special = val === null ? ' (null special case!)' : Array.isArray(val) ? ' (array is an object!)' : '';
      out.textContent = `typeof ${raw} === "${t}"${special}
Value: ${JSON.stringify(val)}
Falsy? ${!val}`;
    } catch(e) { out.textContent = 'Parse error: ' + e.message; }
  }
  const questions = [
    {expr:"'3' + 4", ans:"'34'"},
    {expr:"'3' - 4", ans:"-1"},
    {expr:"true + 1", ans:"2"},
    {expr:"false + false", ans:"0"},
    {expr:"null + 1", ans:"1"},
    {expr:"typeof null", ans:"'object'"},
    {expr:"Boolean('')", ans:"false"},
    {expr:"Number('hello')", ans:"NaN"},
  ];
  let qi=0,score=0,started=false;
  function startQuiz(){
    qi=0;score=0;started=true;
    document.getElementById('q-ans').style.display='block';
    document.getElementById('q-btn').textContent='Submit Answer';
    document.getElementById('q-btn').onclick=submitAnswer;
    showQ();
  }
  function showQ(){
    if(qi>=questions.length){
      document.getElementById('q-text').textContent='Quiz complete!';
      document.getElementById('q-ans').style.display='none';
      document.getElementById('q-btn').textContent='Restart';
      document.getElementById('q-btn').onclick=startQuiz;
      return;

    const ok=ans===correct||ans===correct.replace(/'/g,'');
    if(ok) score++;
    document.getElementById('q-out').textContent=ok?`✅ Correct! ${questions[qi].expr} = ${correct}`:`❌ Answer was: ${correct}. Got: "${ans}"`;
    document.getElementById('q-score').textContent=score;
