Phase 1 of 2
What is JavaScript?
📖 Concept

JavaScript (JS) is the programming language of the web. HTML creates structure, CSS adds style, and JavaScript adds behaviour — things happen when users click, type, or move. JS runs directly in the browser, no installation needed. It's the most-used programming language in the world.

script.js
// This is a JavaScript comment

// JS can change HTML content
document.getElementById('demo').innerHTML = 'Hello, World!';

// JS can change CSS styles
document.getElementById('demo').style.color = '#d97706';

// JS can respond to events
document.getElementById('btn').onclick = function() {
  alert('Button clicked!');
};

// JS can do maths
let result = 2 + 2;
console.log(result); // 4 — open DevTools to see this
✏️ Your Task

Open your browser DevTools (F12 → Console) and type 2 + 2 then press Enter. JavaScript works right in the browser! Also try document.title to read the page title.

Phase 2 of 2
The Role of JavaScript
📖 Concept

JS is the third pillar of web development: HTML = content, CSS = presentation, JS = behaviour. JS can: show/hide elements, validate forms, fetch data from servers, create animations, build entire apps (React, Vue). Modern JS (ES6+) is powerful, readable, and used everywhere from browsers to servers (Node.js).

script.js
// JS changes the DOM (Document Object Model)
// The DOM is the live tree of all HTML elements

// Read an element
let heading = document.querySelector('h1');
console.log(heading.textContent);

// Change an element
heading.textContent = 'Changed by JavaScript!';
heading.style.color = '#d97706';

// Create a new element
let para = document.createElement('p');
para.textContent = 'Injected by JS';
document.body.appendChild(para);

// Remove an element
// para.remove();
✏️ Your Task

Use DevTools Console: type document.querySelector('h1').style.background = 'gold' on any web page. You just hacked a website with JS (only on your screen)!

  *{box-sizing:border-box;margin:0;padding:0}
  body{font-family:Arial,sans-serif;background:#fffbeb;color:#1c1917;min-height:100vh;display:flex;flex-direction:column;align-items:center;justify-content:center;gap:24px;padding:24px}

  .card h2{color:#92400e;font-size:1rem;margin-bottom:16px}
  .btn{display:inline-block;background:#d97706;color:white;padding:10px 24px;border-radius:9px;border:none;font-size:14px;font-weight:700;cursor:pointer;transition:.15s;margin:6px}
  .btn:hover{background:#b45309;transform:translateY(-1px)}
  .count{font-size:3rem;font-weight:800;color:#d97706;margin:12px 0}
  .clock{font-size:1.8rem;font-weight:700;color:#d97706;font-family:monospace}

<body>
  <h1>⚡ JavaScript Introduction</h1>
  <p>JavaScript makes web pages interactive. All three demos below run in the browser — no server needed!</p>

  <div class="card">
    <h2>1️⃣ Counter Button</h2>
    <div class="count" id="count">0</div>
    <button class="btn" onclick="changeCount(1)">+1</button>
    <button class="btn" style="background:#92400e" onclick="changeCount(-1)">-1</button>
    <button class="btn" style="background:#64748b" onclick="resetCount()">Reset</button>
  </div>

  <div class="card">
    <h2>2️⃣ Colour Changer</h2>
    <div class="colour-box" id="colBox"></div>
    <button class="btn" onclick="randColour()">Random Colour</button>
    <p id="colLabel" style="margin-top:8px;font-size:12px;color:#92400e">#d97706</p>
  </div>

  <div class="card">
    <h2>3️⃣ Live Clock</h2>
    <div class="clock" id="clock">--:--:--</div>
    <p style="font-size:12px;color:#92400e;margin-top:8px">Updates every second via setInterval()</p>
  </div>

<script>
  // Counter
  let count = 0;
  function changeCount(n) {
    count += n;
    document.getElementById('count').textContent = count;
  }
  function resetCount() { count = 0; document.getElementById('count').textContent = 0; }

  // Random colour
  function randColour() {
    const hex = '#' + Math.floor(Math.random()*0xFFFFFF).toString(16).padStart(6,'0');
    document.getElementById('colBox').style.background = hex;
    document.getElementById('colLabel').textContent = hex;
  }

  // Live clock
  function tick() {
    document.getElementById('clock').textContent = new Date().toLocaleTimeString();
  }
  setInterval(tick, 1000);
  tick();
</script>
</body>
</html>
