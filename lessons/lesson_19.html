Phase 1 of 2
indexOf, search and match
📖 Concept

indexOf(str, from) — finds first position. lastIndexOf(str) — last position. search(regex) — like indexOf but with regex. match(regex) — returns array of matches (or null). With g flag returns all matches. matchAll(regex) — returns iterator of all matches with groups.

script.js
const text = 'The cat sat on the mat. The cat is fat.';

// indexOf
text.indexOf('cat');      // 4
text.indexOf('cat', 5);   // 27 (search from pos 5)
text.lastIndexOf('cat');  // 27
text.indexOf('dog');      // -1

// search — with regex
text.search(/cat/);       // 4
text.search(/Cat/i);      // 4 (case-insensitive)

// match — first match
text.match(/cat/);        // ['cat', index:4, ...]
text.match(/dog/);        // null

// match — all matches with /g flag
text.match(/cat/g);       // ['cat', 'cat']
text.match(/at/g);        // ['at', 'at', 'at', 'at']

// matchAll — all matches with capture groups
const email = 'Send to amina@test.com or chidi@ng.co';
const emailRx = /[\w.-]+@[\w.-]+\.\w+/g;
const emails = [...email.matchAll(emailRx)];
emails.map(m => m[0]);  // ['amina@test.com', 'chidi@ng.co']

Phase 2 of 2
Regex basics
📖 Concept

Regular expressions match patterns in strings. Create with /pattern/flags or new RegExp(pattern, flags). Flags: g (global), i (case-insensitive), m (multiline). Basics: . = any char, * = 0+, + = 1+, ? = 0 or 1, \d = digit, \w = word char, [abc] = any of a,b,c.

script.js
// Basic patterns
/hello/i.test('Hello World');     // true (case insensitive)
/^Hello/.test('Hello World');     // true (starts with)
/World$/.test('Hello World');     // true (ends with)

// Character classes
/\d+/.test('Room 42');           // true (one or more digits)
/\w+/.test('hello_123');         // true (word chars)
/[aeiou]/g.test('hello');        // true (any vowel)

// Common patterns
const phoneRx  = /^[\d\s+()-]{10,15}$/;
const emailRx  = /^[\w.-]+@[\w.-]+\.\w{2,}$/;
const urlRx    = /^https?:\/\/[\w.-]+/;
const nameRx   = /^[A-Za-z\s'-]{2,50}$/;

// Test email
emailRx.test('amina@example.com');  // true
emailRx.test('not-an-email');        // false

// Replace with regex
'hello world'.replace(/\b\w/g, c => c.toUpperCase()); // 'Hello World'

  function doSearch() {
    const text = document.getElementById('search-text').value;
    const term = document.getElementById('search-term').value;
    if (!term) { document.getElementById('search-display').textContent = text; return; }
    try {
      const isRegex = term.startsWith('/') && term.lastIndexOf('/') > 0;
      let rx;
      if (isRegex) {
        const lastSlash = term.lastIndexOf('/');
        rx = new RegExp(term.slice(1,lastSlash), term.slice(lastSlash+1) || 'gi');
      } else {
        rx = new RegExp(term.replace(/[.*+?^${}()|[\]\]/g,'\$&'), 'gi');
      }
      const matches = text.match(rx) || [];
      document.getElementById('search-stats').textContent =
        `Found ${matches.length} match${matches.length!==1?'es':''} for "${term}"`;
      const highlighted = text.replace(rx, m => `<mark class="highlight">${m}</mark>`);
      document.getElementById('search-display').innerHTML = highlighted;
    } catch(e) {
      document.getElementById('search-stats').textContent = 'Invalid regex: ' + e.message;
      document.getElementById('search-display').textContent = text;
    }
  }
  doSearch();
  const PATTERNS = {
    name:  { rx: /^[A-Za-z\s'-]{2,50}$/, msg: '2–50 letters, spaces, hyphens, apostrophes' },
    email: { rx: /^[\w.+-]+@[\w-]+\.\w{2,}$/, msg: 'user@domain.tld format' },
    phone: { rx: /^[\d\s+()-]{10,15}$/, msg: '10–15 digits with optional +, spaces, dashes' },
  };
  function validateField(el, type) {
    const val = el.value.trim();
    const {rx, msg} = PATTERNS[type];
    const ok = val.length > 0 && rx.test(val);
    const empty = val.length === 0;
    el.className = empty ? '' : ok ? 'valid' : 'invalid';
    const msgEl = document.getElementById('v-' + type + '-msg');
    msgEl.textContent = empty ? '' : ok ? '✅ Valid' : '❌ ' + msg;
    msgEl.className = 'field-msg ' + (ok ? 'ok' : 'err');
    updateSummary();
  }
  function updateSummary() {
    const fields = ['name','email','phone'];
    const results = fields.map(f => {
      const val = document.getElementById('v-'+f).value.trim();
      const ok = val.length > 0 && PATTERNS[f].rx.test(val);
      return `${f}: ${val ? (ok?'✅':'❌') : '⬜'}`;
    });
    document.getElementById('v-summary').textContent = results.join('
');
