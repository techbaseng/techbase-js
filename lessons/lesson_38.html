Phase 1 of 2
for...of basics
📖 Concept

for(const item of iterable) works with any iterable: arrays, strings, Maps, Sets, NodeLists, arguments, generators. Supports break and continue. Use entries() to get index+value: for(const [i,v] of arr.entries()). Cannot iterate plain objects (they are not iterable).

script.js
// Array
for (const fruit of ['mango', 'guava', 'pawpaw']) {
  console.log(fruit);
}

// String (iterates characters)
for (const char of 'hello') {
  console.log(char); // h,e,l,l,o
}

// Map
const map = new Map([['a',1],['b',2]]);
for (const [key, val] of map) {
  console.log(key, val);
}

// Set (unique values)
const set = new Set([1,2,3,2,1]);
for (const val of set) {
  console.log(val); // 1,2,3 (no duplicates)
}

// With index via entries()
const arr = ['a','b','c'];
for (const [i, v] of arr.entries()) {
  console.log(i, v); // 0 a, 1 b, 2 c
}

Phase 2 of 2
Generators and custom iterables
📖 Concept

A generator function (function*) yields values on demand. yield pauses the function and returns a value. Call next() to resume. Generators are iterables — use with for...of. Infinite sequences are possible!

script.js
// Generator function
function* count(start, end) {
  for (let i = start; i <= end; i++) {
    yield i; // pauses here each time
  }
}

for (const n of count(1, 5)) {
  console.log(n); // 1,2,3,4,5
}

// Infinite generator (use break!)
function* naturals() {
  let n = 1;
  while (true) yield n++;
}
const gen = naturals();
for (const n of gen) {
  console.log(n);
  if (n === 5) break; // must break!
}

// Spread a generator
[...count(1, 5)]; // [1,2,3,4,5]
✏️ Your Task

function iterDemo(type){
  let lines=[];
  switch(type){
    case 'array':
      const fruits=['mango','guava','pawpaw','orange'];
      lines.push('for (const fruit of ['+fruits.map(f=>`"${f}"`).join(',')+']) :');
      for(const f of fruits) lines.push('  '+f);
      break;
    case 'string':
      lines.push('for (const char of "Hello") :');
      for(const c of 'Hello') lines.push('  "'+c+'"');
      break;
    case 'map':
      const map=new Map([['NGN','₦'],['USD','$'],['GBP','£'],['EUR','€']]);
      lines.push('for (const [code, symbol] of currencyMap) :');
      for(const[k,v]of map) lines.push(`  ${k} → ${v}`);
      break;
    case 'set':
      const set=new Set([1,2,3,2,1,4,3,5]);
      lines.push('new Set([1,2,3,2,1,4,3,5]) — duplicates removed:');
      lines.push('for (const val of set) :');
      for(const v of set) lines.push('  '+v);
      break;
    case 'entries':
      const arr=['alpha','beta','gamma','delta'];
      lines.push('for (const [i, v] of arr.entries()) :');
      for(const[i,v]of arr.entries()) lines.push(`  [${i}] ${v}`);
      break;
  }
  document.getElementById('iter-out').textContent=lines.join('
');
}
function* fibGen(){
  let[a,b]=[1,1];
  while(true){yield a;[a,b]=[b,a+b];}
}
function fibonacci(){
  const n=Math.min(parseInt(document.getElementById('fib-n').value)||10,30);
  const gen=fibGen(); const nums=[];
  for(const v of gen){nums.push(v);if(nums.length>=n)break;}
  document.getElementById('fib-out').textContent=
    `First ${n} Fibonacci numbers:
[${nums.join(', ')}]

Sum: ${nums.reduce((a,b)=>a+b,0)}
Largest: ${nums[nums.length-1]}`;
}
