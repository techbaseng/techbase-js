Phase 1 of 2
Number instance methods
📖 Concept

toFixed(n) — fixed decimal places (returns string). toPrecision(n) — significant figures. toString(base) — convert to different base (2, 8, 16). toLocaleString(locale, options) — culture-aware formatting. valueOf() — returns primitive value.

script.js
const pi = 3.14159265358979;
const bigNum = 1234567.89;

// toFixed — decimal places
pi.toFixed(0);   // "3"
pi.toFixed(2);   // "3.14"
pi.toFixed(5);   // "3.14159"

// toPrecision — significant figures
pi.toPrecision(4);  // "3.142"
pi.toPrecision(8);  // "3.1415927"

// toString — base conversion
(255).toString(16);  // "ff"
(255).toString(2);   // "11111111"
(255).toString(8);   // "377"
(10).toString(2);    // "1010"

// toLocaleString — cultural formatting
bigNum.toLocaleString();           // "1,234,567.89" (US)
bigNum.toLocaleString('en-NG');    // "1,234,567.89"
bigNum.toLocaleString('de-DE');    // "1.234.567,89" (Germany!)

// Currency formatting
(5000).toLocaleString('en-NG', {
  style: 'currency',
  currency: 'NGN'
}); // "NGN 5,000.00" or "₦5,000.00"

// Compact notation
(1000000).toLocaleString('en', {
  notation: 'compact'
}); // "1M

Phase 2 of 2
Static Number methods
📖 Concept

Number.isInteger(n), Number.isFinite(n), Number.isNaN(n), Number.isSafeInteger(n), Number.parseInt(str), Number.parseFloat(str). Prefer the static methods over global — they don't do type coercion.

script.js
// Static methods — more strict than global versions
Number.isNaN(NaN);          // true
Number.isNaN('hello');      // false (no coercion!)
isNaN('hello');             // true  (coerces to NaN first)

Number.isFinite(42);        // true
Number.isFinite(Infinity);  // false
Number.isFinite('42');      // false (no coercion!)
isFinite('42');             // true  (coerces '42' to 42)

Number.isInteger(42);       // true
Number.isInteger(42.5);     // false
Number.isInteger(42.0);     // true!

Number.isSafeInteger(9007199254740991);   // true
Number.isSafeInteger(9007199254740992);   // false

Number.parseInt('42px', 10);   // 42
Number.parseFloat('3.14em');   // 3.14

// Parsing edge cases
parseInt('08', 10);   // 8  (always specify base 10!)
parseInt('0x1F', 16); // 31 (hex prefix)
parseFloat('.5');     // 0.5
