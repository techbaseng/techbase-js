Phase 1 of 2
Searching and extracting
📖 Concept

indexOf(str) — first occurrence index (-1 if not found). lastIndexOf() — last occurrence. includes(str) — true/false. startsWith() / endsWith(). slice(start, end) — extract substring (negative counts from end). substring(start, end) — like slice but no negatives. at(index) — supports negative indices.

script.js
const str = 'Techbase Academy — JavaScript Course';

// Searching
str.indexOf('Academy');      // 10
str.indexOf('Python');       // -1 (not found)
str.includes('JavaScript');  // true
str.startsWith('Tech');      // true
str.endsWith('Course');      // true

// Extracting
str.slice(10, 17);     // 'Academy'
str.slice(-6);         // 'Course' (from end)
str.slice(0, 9);       // 'Techbase'
str.at(0);             // 'T'
str.at(-1);            // 'e' (last character)
str.substring(0, 9);   // 'Techbase'

// Splitting
'apple,banana,mango'.split(',');    // ['apple','banana','mango']
'hello world'.split(' ');           // ['hello', 'world']
'hello'.split('');                  // ['h','e','l','l','o']
'a.b.c'.split('.', 2);             // ['a', 'b'] (limit)

Phase 2 of 2
Transforming and cleaning
📖 Concept

toUpperCase() / toLowerCase(). trim() — removes whitespace from both ends. trimStart() / trimEnd(). replace(search, replacement) — first match only. replaceAll() — all matches. padStart(length, char) / padEnd(). repeat(n).

script.js
// Transforming
'hello'.toUpperCase();       // 'HELLO'
'WORLD'.toLowerCase();       // 'world'

// Cleaning
'  hello  '.trim();          // 'hello'
'  hello  '.trimStart();     // 'hello  '
'  hello  '.trimEnd();       // '  hello'

// Replacing
'cat cat cat'.replace('cat', 'dog');    // 'dog cat cat'
'cat cat cat'.replaceAll('cat', 'dog'); // 'dog dog dog'
'hello'.replace(/l/g, 'r');            // 'herro' (regex global)

// Padding
'5'.padStart(3, '0');   // '005' (like '007')
'hi'.padEnd(5, '!');    // 'hi!!!'
'42'.padStart(4);       // '  42' (space padding)

// Chaining
const result = '  Hello World  '
  .trim()
  .toLowerCase()
  .replace('hello', 'Hi')
  .split(' ')
  .join('-');
// 'hi-world'

  function inspectMethods() {
    const str = document.getElementById('str-in').value;
    const container = document.getElementById('methods-out');
    const methods = [
      {name:'.length',       val: str.length},
      {name:'.toUpperCase()', val: str.toUpperCase()},
      {name:'.toLowerCase()', val: str.toLowerCase()},
      {name:'.trim()',        val: str.trim()},

  function findReplace(all) {

    const find = document.getElementById('fr-find').value;

(${count} occurrence${count!==1?'s':''} found)`;

    const email = document.getElementById('email-in').value;
    const hasAt  = email.includes('@');

    const noSpace = !email.includes(' ');
